\documentclass{article}

\usepackage{lipsum}

% Define a box for an element to be focused
\makeatletter
\newsavebox{\@elementbox}
\newcommand{\elementbox}[2]{%
    \savebox{\@elementbox}{\vbox{#2}}%
    \pdfsavepos
    \usebox{\@elementbox}%
    \directlua{
        add_element(
            '#1',
            tex.count.c@page,
            tex.sp(tex.pdflastxpos),
            tex.sp(tex.pdflastypos),
            tex.sp(\the\wd\@elementbox),
            tex.sp(\the\ht\@elementbox)
        )
    }%
}
\makeatother

% Define a command to add narration steps
\newcommand{\narration}[2]{\directlua{add_narration_step("#1", [[#2]])}}
